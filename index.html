let chartAcelInstance = null; let chartFFTInstance = null; let tamanhoUltimaLeitura = 0; let isFetching = false; function parseNum(val) { if (typeof val === 'number') return val; if (!val) return 0; return parseFloat(String(val).replace(',', '.')) || 0; } // Coloque aqui o URL do Apps Script que criou no Passo 1 const WEB_APP_URL = "SUA_URL_DO_APPS_SCRIPT_AQUI"; async function carregarDados() { if (isFetching) return; let url = document.getElementById('csvUrl').value.trim(); if (!url || url.includes('docs.google.com/spreadsheets')) { url = WEB_APP_URL; } isFetching = true; try { const response = await fetch(url + (url.includes('?') ? '&' : '?') + '_t=' + Date.now()); const data = await response.json(); if (Array.isArray(data) && data.length > 0) { processarDados(data); document.getElementById('lastUpdate').innerText = 'Atualizado às ' + new Date().toLocaleTimeString(); } } catch (err) { console.error('Erro ao procurar novos dados:', err); } finally { isFetching = false; } } function processarDados(data) { const quantidadeAtual = data.length; if (quantidadeAtual !== tamanhoUltimaLeitura) { tamanhoUltimaLeitura = quantidadeAtual; let tempos = []; let acelX = []; let acelY = []; let acelZ = []; let maxZ = 0; let ultimoStatus = "Normal"; data.forEach(row => { let t = parseNum(row['Tempo (s)'] || row['Tempo'] || 0); let x = parseNum(row['Aceleração X (g)'] || row['Aceleração X'] || row['Aceleraçao X (g)'] || 0); let y = parseNum(row['Aceleração Y (g)'] || row['Aceleração Y'] || row['Aceleraçao Y (g)'] || 0); let z = parseNum(row['Aceleração Z (g)'] || row['Aceleração Z'] || row['Aceleraçao Z (g)'] || 0); tempos.push(t); acelX.push(x); acelY.push(y); acelZ.push(z); if (Math.abs(z) > maxZ) maxZ = Math.abs(z); if (row['Status'] || row['Status Estrutural']) { ultimoStatus = row['Status'] || row['Status Estrutural']; } }); document.getElementById('kpiStatus').innerText = ultimoStatus; document.getElementById('kpiMaxZ').innerText = maxZ.toFixed(4); document.getElementById('kpiLeituras').innerText = quantidadeAtual.toLocaleString('pt-BR'); chartAcelInstance.data.labels = tempos; chartAcelInstance.data.datasets[0].data = acelX; chartAcelInstance.data.datasets[1].data = acelY; chartAcelInstance.data.datasets[2].data = acelZ; chartAcelInstance.update('none'); } } // Verificação a cada 1 segundo setInterval(carregarDados, 1000);
